cmake_minimum_required(VERSION 3.28)
project(excel-interactio VERSION 0.1.1 LANGUAGES CXX)
set(INTERACTIO_VERSION "0.1.1-beta.1" CACHE STRING "Release version, optionally with -beta.N")
if(NOT INTERACTIO_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)(-beta\\.[1-9][0-9]*)?$")
  message(FATAL_ERROR "INTERACTIO_VERSION must be X.Y.Z or X.Y.Z-beta.N")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(EXCEL_OBS_DEPS "" CACHE PATH "Prepared OBS development dependencies")
if(NOT EXCEL_OBS_DEPS)
  message(FATAL_ERROR "Set EXCEL_OBS_DEPS to the prepared OBS dependencies")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${EXCEL_OBS_DEPS}/qt6")
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network)
foreach(name IN ITEMS obs obs-frontend-api)
  add_library(${name} SHARED IMPORTED GLOBAL)
  set_target_properties(${name} PROPERTIES
    IMPORTED_IMPLIB "${EXCEL_OBS_DEPS}/obs-sdk/${name}.lib"
    IMPORTED_LOCATION "${EXCEL_OBS_DEPS}/obs-runtime/bin/64bit/${name}.dll"
    INTERFACE_INCLUDE_DIRECTORIES "${EXCEL_OBS_DEPS}/obs-studio/libobs;${EXCEL_OBS_DEPS}/obs-studio/frontend/api;${EXCEL_OBS_DEPS}/obs-sdk")
endforeach()
add_library(excel-interactio MODULE src/plugin.cpp src/remote-client.cpp src/update-controller.cpp src/update-policy.cpp)
target_link_libraries(excel-interactio PRIVATE obs obs-frontend-api Qt6::Core Qt6::Widgets Qt6::Network advapi32)
target_compile_definitions(excel-interactio PRIVATE INTERACTIO_VERSION="${INTERACTIO_VERSION}" FX_VERSION="${INTERACTIO_VERSION}" INTERACTIO_GITHUB_REPOSITORY="StayExcellent666/ExcelInteractio")
target_compile_options(excel-interactio PRIVATE /utf-8 /EHsc)
set_target_properties(excel-interactio PROPERTIES PREFIX "")
install(TARGETS excel-interactio RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
install(FILES "${EXCEL_OBS_DEPS}/qt6/plugins/tls/qschannelbackend.dll" DESTINATION data/obs-plugins/excel-interactio/tls)
install(DIRECTORY data/overlays DESTINATION data/obs-plugins/excel-interactio)
install(FILES LICENSE DESTINATION data/obs-plugins/excel-interactio)
add_executable(excel-interactio-updater WIN32 updater/main.cpp)
target_link_libraries(excel-interactio-updater PRIVATE bcrypt shell32)
target_compile_definitions(excel-interactio-updater PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(excel-interactio-updater PRIVATE /utf-8 /EHsc)
set_property(TARGET excel-interactio-updater PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
install(TARGETS excel-interactio-updater RUNTIME DESTINATION data/obs-plugins/excel-interactio)
add_executable(interactio-update-policy-test tests/update-policy-test.cpp src/update-policy.cpp)
target_link_libraries(interactio-update-policy-test PRIVATE Qt6::Core)
target_include_directories(interactio-update-policy-test PRIVATE src)
target_compile_options(interactio-update-policy-test PRIVATE /utf-8 /EHsc)
add_executable(interactio-update-e2e-launcher tests/update-e2e-launcher.cpp)
target_compile_definitions(interactio-update-e2e-launcher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(interactio-update-e2e-launcher PRIVATE /utf-8 /EHsc)
add_executable(interactio-placement-test tests/placement-test.cpp src/remote-client.cpp src/update-controller.cpp src/update-policy.cpp)
target_link_libraries(interactio-placement-test PRIVATE obs obs-frontend-api Qt6::Core Qt6::Widgets Qt6::Network advapi32)
target_compile_definitions(interactio-placement-test PRIVATE INTERACTIO_VERSION="0.1.0" INTERACTIO_GITHUB_REPOSITORY="StayExcellent666/ExcelInteractio")
target_compile_options(interactio-placement-test PRIVATE /utf-8 /EHsc)
add_executable(interactio-ui-preview tests/ui-preview.cpp src/remote-client.cpp src/update-controller.cpp src/update-policy.cpp)
target_link_libraries(interactio-ui-preview PRIVATE obs obs-frontend-api Qt6::Core Qt6::Widgets Qt6::Network advapi32)
target_compile_definitions(interactio-ui-preview PRIVATE INTERACTIO_VERSION="0.1.0" INTERACTIO_GITHUB_REPOSITORY="StayExcellent666/ExcelInteractio")
target_compile_options(interactio-ui-preview PRIVATE /utf-8 /EHsc)
